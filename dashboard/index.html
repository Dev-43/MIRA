from playwright.sync_api import sync_playwright

def test_drone1_flight_state_reconciliation():

    state = requests.get("http://127.0.0.1:4000/api/control/state").json()
    gt_status = state["drones"]["drone-1"]["status"]

    # 2. Inspect Cockpit UI Rendered Row
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=True)
        page = browser.new_page()
        page.goto("http://127.0.0.1:4010")
        row_text = page.locator(".device-row").first.inner_text()
        browser.close()

    # 3. Assert Ground Truth vs UI State Fidelity
    if gt_status == "in_flight":
        assert "in_flight" in row_text, f"State Desync: Simulator is in_flight, but UI displays: {row_text}"
